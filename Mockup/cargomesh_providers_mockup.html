document.modelContext.registerTool({
  name: "quote_freight",
  description: "Generates an official guaranteed freight quote for Andes Freight",
  inputSchema: {
    type: "object",
    properties: {
      origin: { type: "string" },
      destination: { type: "string" },
      cargo_weight_kg: { type: "number" }
    },
    required: ["origin", "destination", "cargo_weight_kg"]
  },
  execute: async (input) => {
    return {
      price: 760,
      currency: "USD",
      estimated_duration_hours: 16,
      assigned_vehicle: "Scania R450 (18,000 kg max)",
      valid_until: "2026-08-31T18:00:00Z"
    };
  }
});
